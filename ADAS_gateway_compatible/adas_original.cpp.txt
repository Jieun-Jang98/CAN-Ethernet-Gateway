
#include <algorithm>
#include <chrono>
#include <cmath>
#include <csignal>
#include <cstdint>
#include <cerrno>
#include <cstring>
#include <iomanip>
#include <iostream>
#include <stdexcept>
#include <string>

#include <arpa/inet.h>
#include <sys/socket.h>
#include <unistd.h>

using Clock = std::chrono::steady_clock;

constexpr const char* GATEWAY_IP = "192.168.50.10";
constexpr uint16_t ADAS_LISTEN_PORT = 5000;
constexpr uint16_t GATEWAY_COMMAND_PORT = 5001;

constexpr uint8_t STATE_TYPE = 0x01;
constexpr uint8_t COMMAND_TYPE = 0x02;
constexpr size_t STATE_SIZE = 11;
constexpr size_t COMMAND_SIZE = 6;

constexpr double KP = 0.5;
constexpr double MAX_TORQUE_SLEW = 10.0; // 모의 단위/s
constexpr auto STATE_TIMEOUT = std::chrono::milliseconds(100);
constexpr auto FAILSAFE_PERIOD = std::chrono::milliseconds(20);

volatile std::sig_atomic_t running = 1;

void stopHandler(int) {
    running = 0;
}

double clampValue(double x, double low, double high) {
    return std::max(low, std::min(x, high));
}

uint16_t readU16(const uint8_t* p) {
    return (uint16_t(p[0]) << 8) | uint16_t(p[1]);
}

int16_t readI16(const uint8_t* p) {
    return static_cast<int16_t>(readU16(p));
}

void writeI16(uint8_t* p, int16_t value) {
    uint16_t u = static_cast<uint16_t>(value);
    p[0] = static_cast<uint8_t>(u >> 8);
    p[1] = static_cast<uint8_t>(u & 0xFF);
}

struct VehicleState {
    uint8_t counter;
    double speedFL;
    double speedFR;
    double angle;
    double driverTorque;
};

struct Command {
    bool enable;
    double torque;
    std::string status;
};

bool decodeState(const uint8_t* data, size_t size,
                 VehicleState& state) {
    if (size != STATE_SIZE) return false;
    if (data[0] != STATE_TYPE || data[2] != STATE_SIZE)
        return false;

    state.counter = data[1];
    state.speedFL = readU16(data + 3) / 100.0;
    state.speedFR = readU16(data + 5) / 100.0;
    state.angle = readI16(data + 7) / 10.0;
    state.driverTorque = readI16(data + 9);

    return true;
}

Command calculateCommand(const VehicleState& s,
                         double previousTorque,
                         double dt) {
    if (!std::isfinite(s.speedFL) ||
        !std::isfinite(s.speedFR) ||
        !std::isfinite(s.angle) ||
        !std::isfinite(s.driverTorque)) {
        return {false, 0.0, "INVALID_VALUE"};
    }

    if (s.speedFL < 0 || s.speedFL > 250 ||
        s.speedFR < 0 || s.speedFR > 250) {
        return {false, 0.0, "INVALID_SPEED"};
    }

    if (std::abs(s.speedFL - s.speedFR) > 15.0) {
        return {false, 0.0, "WHEEL_SPEED_MISMATCH"};
    }

    double speed = (s.speedFL + s.speedFR) / 2.0;

    double angleLimit;
    double torqueLimit;

    if (speed < 30.0) {
        angleLimit = 30.0;
        torqueLimit = 10.0;
    } else if (speed < 80.0) {
        angleLimit = 10.0;
        torqueLimit = 5.0;
    } else {
        angleLimit = 3.0;
        torqueLimit = 2.0;
    }

    if (std::abs(s.angle) > angleLimit) {
        return {false, 0.0, "ANGLE_OUT_OF_RANGE"};
    }

    if (std::abs(s.driverTorque) > torqueLimit) {
        return {false, 0.0, "DRIVER_OVERRIDE"};
    }

    double targetTorque =
        clampValue(-KP * s.angle, -torqueLimit, torqueLimit);

    double maxStep = MAX_TORQUE_SLEW * std::max(0.0, dt);

    double requestedTorque = clampValue(
        targetTorque,
        previousTorque - maxStep,
        previousTorque + maxStep
    );

    return {true, requestedTorque, "ACTIVE"};
}

int main() {
    std::signal(SIGINT, stopHandler);
    std::signal(SIGTERM, stopHandler);

    int sock = socket(AF_INET, SOCK_DGRAM, 0);
    if (sock < 0) {
        std::cerr << "socket error: " << std::strerror(errno) << "\n";
        return 1;
    }

    sockaddr_in localAddr{};
    localAddr.sin_family = AF_INET;
    localAddr.sin_port = htons(ADAS_LISTEN_PORT);
    localAddr.sin_addr.s_addr = htonl(INADDR_ANY);

    if (bind(sock, reinterpret_cast<sockaddr*>(&localAddr),
             sizeof(localAddr)) < 0) {
        std::cerr << "bind error: " << std::strerror(errno) << "\n";
        close(sock);
        return 1;
    }

    sockaddr_in gatewayAddr{};
    gatewayAddr.sin_family = AF_INET;
    gatewayAddr.sin_port = htons(GATEWAY_COMMAND_PORT);

    if (inet_pton(AF_INET, GATEWAY_IP,
                  &gatewayAddr.sin_addr) != 1) {
        std::cerr << "Invalid Gateway IP\n";
        close(sock);
        return 1;
    }

    timeval timeout{};
    timeout.tv_sec = 0;
    timeout.tv_usec = 10000; // 10ms
    setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO,
               &timeout, sizeof(timeout));

    uint8_t commandCounter = 0;
    int lastStateCounter = -1;
    double lastTorque = 0.0;

    auto now = Clock::now();
    auto lastValidTime = now;
    auto lastControlTime = now;
    auto lastFailsafeTime = now - FAILSAFE_PERIOD;

    bool haveValidState = false;
    bool timedOut = true;

    auto sendCommand = & {
        uint8_t packet[COMMAND_SIZE]{};

        packet[0] = COMMAND_TYPE;
        packet[1] = commandCounter++;
        packet[2] = COMMAND_SIZE;
        packet[3] = enable ? 1 : 0;

        double limited =
            clampValue(torque, -327.68, 327.67);
        int16_t raw = static_cast<int16_t>(
            std::round(limited * 100.0)
        );

        writeI16(packet + 4, raw);

        ssize_t sent = sendto(
            sock, packet, sizeof(packet), 0,
            reinterpret_cast<sockaddr*>(&gatewayAddr),
            sizeof(gatewayAddr)
        );

        if (sent != static_cast<ssize_t>(sizeof(packet))) {
            std::cerr << "UDP send error: "
                      << std::strerror(errno) << "\n";
        }
    };

    std::cout << "ADAS Controller started\n";
    std::cout << "Listen UDP: " << ADAS_LISTEN_PORT << "\n";
    std::cout << "Gateway: " << GATEWAY_IP << ":"
              << GATEWAY_COMMAND_PORT << "\n";

    while (running) {
        uint8_t buffer[1024]{};

        sockaddr_in sender{};
        socklen_t senderLen = sizeof(sender);

        ssize_t received = recvfrom(
            sock, buffer, sizeof(buffer), 0,
            reinterpret_cast<sockaddr*>(&sender),
            &senderLen
        );

        now = Clock::now();

        // 유효한 차량 상태가 100ms 이상 없으면 비활성 명령 반복
        if (!haveValidState ||
            now - lastValidTime >= STATE_TIMEOUT) {

            if (!timedOut) {
                std::cout << "TIMEOUT: ADAS disabled\n";
            }

            timedOut = true;
            lastTorque = 0.0;

            if (now - lastFailsafeTime >= FAILSAFE_PERIOD) {
                sendCommand(false, 0.0);
                lastFailsafeTime = now;
            }
        }

        if (received < 0) {
            if (errno == EAGAIN || errno == EWOULDBLOCK ||
                errno == EINTR) {
                continue;
            }

            std::cerr << "recvfrom error: "
                      << std::strerror(errno) << "\n";
            continue;
        }

        // 지정된 Gateway IP에서 온 데이터만 수신
        if (sender.sin_addr.s_addr != gatewayAddr.sin_addr.s_addr) {
            continue;
        }

        VehicleState state{};

        if (!decodeState(buffer,
                         static_cast<size_t>(received), state)) {
            continue;
        }

        if (haveValidState && !timedOut) {
            int delta =
                (int(state.counter) - lastStateCounter + 256) % 256;

            if (delta == 0 || delta > 127) {
                std::cout << "DUPLICATE/OLD COUNTER: "
                          << int(state.counter) << "\n";
                continue;
            }

            if (delta > 1) {
                std::cout << "COUNTER GAP: "
                          << delta - 1 << "\n";
            }
        }

        double dt =
            std::chrono::duration<double>(now - lastControlTime)
                .count();

        // 타임아웃 후 재개 시 오래된 dt가 적용되지 않도록 함
        if (timedOut) dt = 0.02;

        lastStateCounter = state.counter;
        lastValidTime = now;
        lastControlTime = now;
        haveValidState = true;
        timedOut = false;

        Command command =
            calculateCommand(state, lastTorque, dt);

        lastTorque = command.enable ? command.torque : 0.0;

        sendCommand(command.enable, lastTorque);

        std::cout << std::fixed << std::setprecision(2)
                  << "Speed="
                  << (state.speedFL + state.speedFR) / 2.0
                  << " km/h | Angle=" << state.angle
                  << " deg | DriverTq=" << state.driverTorque
                  << " | Enable=" << command.enable
                  << " | RequestTq=" << lastTorque
                  << " | Status=" << command.status << "\n";
    }

    sendCommand(false, 0.0);
    close(sock);

    std::cout << "ADAS Controller stopped\n";
    return 0;
}
