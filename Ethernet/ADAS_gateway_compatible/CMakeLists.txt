cmake_minimum_required(VERSION 3.16)
project(adas_gateway_compatible LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(adas_controller adas_controller.cpp)
add_executable(test_compatibility test_compatibility.cpp gateway_reference/protocol.cpp)
foreach(target adas_controller test_compatibility)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(WIN32)
  target_link_libraries(adas_controller PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME compatibility COMMAND test_compatibility)
